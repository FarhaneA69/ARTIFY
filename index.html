const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req) => { // ============================== // CORS // ============================== if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } try { // ============================== // VERIFICATION METHODE // ============================== if (req.method !== "POST") { return new Response( JSON.stringify({ error: "Méthode non autorisée", }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ============================== // RECUPERATION DES DONNEES // ============================== const body = await req.json(); const imageBase64 = body.image; const prompt = body.prompt; // ============================== // VERIFICATIONS // ============================== if (!imageBase64) { return new Response( JSON.stringify({ error: "Aucune image reçue", }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } if (!prompt || prompt.trim().length === 0) { return new Response( JSON.stringify({ error: "Veuillez écrire une description", }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ============================== // CLE OPENAI // ============================== const OPENAI_API_KEY = Deno.env.get("OPENAI_API_KEY"); if (!OPENAI_API_KEY) { console.error("OPENAI_API_KEY manquante"); return new Response( JSON.stringify({ error: "La clé OPENAI_API_KEY est absente dans Supabase.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ============================== // CONVERSION BASE64 EN IMAGE // ============================== const matches = imageBase64.match( /^data:(image\/[a-zA-Z0-9.+-]+);base64,(.+)$/ ); if (!matches) { return new Response( JSON.stringify({ error: "Format de l'image invalide", }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } const mimeType = matches[1]; const base64Data = matches[2]; const binaryString = atob(base64Data); const bytes = new Uint8Array( binaryString.length ); for (let i = 0; i < binaryString.length; i++) { bytes[i] = binaryString.charCodeAt(i); } const imageBlob = new Blob( [bytes], { type: mimeType, } ); // ============================== // FORM DATA OPENAI // ============================== const formData = new FormData(); formData.append( "image", imageBlob, "photo.png" ); // ============================== // PROMPT FINAL // ============================== const finalPrompt = ` Transform the uploaded photograph according to the user's request. IMPORTANT: - Preserve the main subject and recognizable composition of the original photo. - Transform the image artistically according to the user's instructions. - Create a high-quality, visually impressive result. - Do not simply add a color filter. - Actually transform the visual style, atmosphere, textures, lighting and artistic appearance. - Keep the result suitable for printing as wall art. USER REQUEST: ${prompt} `; formData.append( "prompt", finalPrompt ); formData.append( "model", "gpt-image-1" ); // 3 résultats formData.append( "n", "3" ); formData.append( "size", "1024x1024" ); // ============================== // APPEL OPENAI // ============================== console.log( "Envoi de la demande à OpenAI..." ); const openaiResponse = await fetch( "https://api.openai.com/v1/images/edits", { method: "POST", headers: { Authorization: `Bearer ${OPENAI_API_KEY}`, }, body: formData, } ); const openaiData = await openaiResponse.json(); // ============================== // ERREUR OPENAI // ============================== if (!openaiResponse.ok) { console.error( "Erreur OpenAI:", JSON.stringify(openaiData) ); return new Response( JSON.stringify({ error: openaiData?.error?.message || "Erreur lors de la génération IA", }), { status: openaiResponse.status, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ============================== // RECUPERATION DES IMAGES // ============================== const images = openaiData.data ?.map((item: any) => { if (item.b64_json) { return ( "data:image/png;base64," + item.b64_json ); } if (item.url) { return item.url; } return null; }) .filter(Boolean); if (!images || images.length === 0) { console.error( "Aucune image retournée:", JSON.stringify(openaiData) ); return new Response( JSON.stringify({ error: "L'IA n'a retourné aucune image.", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } // ============================== // REPONSE AU SITE // ============================== console.log( `${images.length} image(s) générée(s)` ); return new Response( JSON.stringify({ success: true, images: images, message: "Images générées avec succès", }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } catch (error) { console.error( "Erreur fonction:", error ); return new Response( JSON.stringify({ error: error?.message || "Une erreur inconnue est survenue", }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } });
